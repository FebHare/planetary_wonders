#Bureaucrat
triggered_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			$OWNER_CONDITION$
			is_regular_empire = yes
			has_technology = pw_tech_galactic_model_expanding_bureaucracy
		}
	}
	modifier = {
		job_bureaucrat_per_pop = $AMOUNT$
	}
}

	}

}
